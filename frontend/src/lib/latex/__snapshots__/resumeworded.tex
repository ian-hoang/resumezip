% Custom commands

% The name in capitals, and the contact line under it
\newenvironment{resumeHeader}{\centering\lineskiplimit=-\maxdimen}{\par\vspace{-3.5bp}}
% An empty box takes the page's first line, so the name's capitals sit at the
% top margin however tall its accents are.
\newcommand{\resumeName}[1]{\hbox{}\vspace{-12.8bp}{\resumeNameFont\textcolor{resumeDark}{\MakeUppercase{#1}}\par}}
\newcommand{\resumeContacts}[1]{\vspace{0.6bp}#1\par}

% A heading in capitals over a dark rule, kept with what follows it.
\newcommand{\resumeSection}[1]{%
  \par\vspace{12.07bp}%
  {\resumeHeadingFont\textcolor{resumeDark}{\MakeUppercase{#1}}\par}%
  \nobreak\vskip\dimexpr3.6bp-\prevdepth\relax
  {\color{resumeRule}\hrule height 0.8pt}%
  % The next line's capitals sit 10.38bp under the rule.
  \nobreak\prevdepth=-4.2bp
}

% A line with one part flush left and one flush right. A long left part
% wraps short of the right one, which sits on its last line.
\newlength{\resumeRightWidth}
\newcommand{\resumeRow}[2]{%
  \par\settowidth{\resumeRightWidth}{#2}%
  {\rightskip=\dimexpr\resumeRightWidth+0.8em\relax plus 1fil\hyphenpenalty=10000\interlinepenalty=10000
    #1\nobreak\hfill\rlap{\hspace{0.8em}#2}\par}%
}
% An organization in gray capitals and its place, then the line under it
\newcommand{\resumeOrganization}[2]{\resumeRow{\textbf{\textcolor{resumeGray}{\MakeUppercase{#1}}}}{#2}}
\newcommand{\resumeSubRow}[2]{\nobreak\resumeRow{#1}{#2}}

% Round bullets at the margin, drawn: EB Garamond's own is small.
\newcommand{\resumeBullet}{%
  \raisebox{1.22bp}{\setlength{\unitlength}{1.8bp}\begin{picture}(2,2)\put(1,1){\circle*{2}}\end{picture}}%
}
\newenvironment{resumeItems}{%
  \nobreak\begin{itemize}[label=\resumeBullet, align=left, labelindent=0pt, labelsep=10.7pt, leftmargin=*]%
}{\end{itemize}\vspace{3.3bp}}

% Paragraphs, as the summary's, 0.6em further apart than lines
\newenvironment{resumeText}{%
  \begin{itemize}[label={}, leftmargin=0pt, labelwidth=0pt, labelsep=0pt, parsep=6.48bp]\item\relax
}{\end{itemize}}
